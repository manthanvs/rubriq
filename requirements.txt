# RubriQ dependencies.
#
# This file grows one phase at a time — see §10. Nothing for Phase 5 is
# installed while Phase 2 is being built, so a broken AI dependency can never
# stop the deterministic core from running.

# --- Phase 0: foundation ---
streamlit>=1.42            # view layer; 1.42 is the floor for native st.login() OIDC
Authlib>=1.3.2             # st.login() needs it and does not pull it in. Without
                           # this the sign-in button raises at runtime, which is a
                           # bad way to discover a missing dependency
SQLAlchemy>=2.0            # engine and ORM; imported only inside core/db
                           # SQLite driver is sqlite3, in the standard library
alembic>=1.13              # migration history is an SDLC deliverable, not optional
python-dotenv>=1.0         # lets Alembic and pytest read .env; they have no st.secrets

# --- Phase 3: text extraction at upload ---
pdfplumber>=0.11           # PDF text; degrades with a note if a scan has none
python-docx>=1.1           # DOCX text, including requirement tables

# --- Phase 4: export ---
openpyxl>=3.1              # a real .xlsx, not a CSV rename (section 3)

# --- Phase 5a: AI response contract and evidence guard ---
pydantic>=2                # enforces the section 6.5 response contract; the
                           # NO_EVIDENCE-scores-zero rule is a validator so no
                           # caller can route around it (fix item 5)
rapidfuzz>=3               # partial_ratio for the anti-hallucination guard

# --- Dev ---
pytest>=8                  # tests/core mirrors core/
ruff>=0.6                  # lint + import order, one tool
Pillow>=10                 # scripts/*_diagram*.py draw the report figures. It
                           # arrives anyway as a pdfplumber dependency, which is
                           # exactly why it is declared here: a transitive
                           # dependency that one of our own modules imports
                           # directly is a build that breaks on somebody else's
                           # release note. Not imported by core/ or app/

google-genai>=1            # decision #4 — Gemini. Imported lazily inside
                           # core/ai/provider.py, so a missing key or SDK is a
                           # readable message rather than a startup failure

openai>=1                  # not OpenAI the vendor — the OpenAI-shaped client,
                           # used to reach OpenRouter (decision #4). One key,
                           # many vendors, including models served free.
                           # Imported lazily inside core/ai/provider.py

# --- Phase 5b: orchestration ---
langgraph>=0.2             # the evaluation graph; imported only inside core/ai
langgraph-checkpoint-sqlite>=2  # SqliteSaver — resumes a run after a Streamlit
                           # rerun instead of re-calling the model (section 6.2)
